package org.example;

import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class udp_data_1 {

    public static void main(String[] args) {

        String serverIP = "36.50.135.242";
        int serverPort = 2207;

        String studentCode = "B22DCCN200";
        String qCode = "Z2pgNquR";

        try {
            DatagramSocket socket = new DatagramSocket();
            InetAddress serverAddress = InetAddress.getByName(serverIP);

            // BƯỚC 1: Gửi mã sinh viên + mã câu hỏi
            String request = ";" + studentCode + ";" + qCode;

            byte[] sendData = request.getBytes(StandardCharsets.UTF_8);

            DatagramPacket sendPacket = new DatagramPacket(
                    sendData,
                    sendData.length,
                    serverAddress,
                    serverPort
            );

            socket.send(sendPacket);

            System.out.println("Đã gửi: " + request);

            // BƯỚC 2: Nhận dữ liệu từ server

            byte[] receiveData = new byte[4096];

            DatagramPacket receivePacket = new DatagramPacket(
                    receiveData,
                    receiveData.length
            );

            socket.receive(receivePacket);

            String response = new String(
                    receivePacket.getData(),
                    0,
                    receivePacket.getLength(),
                    StandardCharsets.UTF_8
            );

            System.out.println("Server gửi: " + response);

            // BƯỚC 3: Phân tích dữ liệu

            String[] parts = response.split(";");

            String requestId = parts[0];
            int n = Integer.parseInt(parts[1]);

            String[] numbers = parts[2].split(",");

            // BƯỚC 4: Đánh dấu các số đã xuất hiện
            boolean[] appeared = new boolean[n + 1];

            for (String number : numbers) {

                int value = Integer.parseInt(number);

                if (value >= 1 && value <= n) {
                    appeared[value] = true;
                }
            }

            // BƯỚC 5: Tìm các số còn thiếu

            List<Integer> missing = new ArrayList<>();

            for (int i = 1; i <= n; i++) {

                if (!appeared[i]) {
                    missing.add(i);
                }
            }
            // BƯỚC 6: Tạo chuỗi kết quả
            StringBuilder result = new StringBuilder();

            result.append(requestId);
            result.append(";");

            for (int i = 0; i < missing.size(); i++) {

                result.append(missing.get(i));

                if (i < missing.size() - 1) {
                    result.append(",");
                }
            }

            String answer = result.toString();

            // BƯỚC 7: Gửi kết quả về server

            byte[] answerData = answer.getBytes(StandardCharsets.UTF_8);

            DatagramPacket answerPacket = new DatagramPacket(
                    answerData,
                    answerData.length,
                    receivePacket.getAddress(),
                    receivePacket.getPort()
            );

            socket.send(answerPacket);

            System.out.println("Đã gửi kết quả: " + answer);

            // BƯỚC 8: Đóng socket

            socket.close();

            System.out.println("Đã đóng socket.");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

            System.out.println("Đã gửi: " + request);

            // requestId;a1,a2,...,a50

            DatagramPacket receivePacket =
                    new DatagramPacket(
                            receiveData,
                            receiveData.length
                    );

            socket.receive(receivePacket);

            String response =
                    new String(
                            receivePacket.getData(),
                            0,
                            receivePacket.getLength()
                    );

            System.out.println("Server gửi: " + response);

            // ==============================
            // 4. Tách requestId và danh sách số
            // ==============================

            String[] parts = response.split(";");

            String requestId = parts[0];

            String[] numberStrings =
                    parts[1].split(",");

            // ==============================
            // 5. Tìm MAX và MIN
            // ==============================

            int max = Integer.MIN_VALUE;
            int min = Integer.MAX_VALUE;

            for (String s : numberStrings) {

                int value = Integer.parseInt(s.trim());

                if (value > max) {
                    max = value;
                }

                if (value < min) {
                    min = value;
                }
            }

            System.out.println("Max = " + max);
            System.out.println("Min = " + min);

            // ==============================
            // 6. Tạo kết quả
            // Format:
            // requestId;max,min
            // ==============================

            String answer =
                    requestId + ";" + max + "," + min;

            // ==============================
            // 7. Gửi kết quả về server
            // ==============================

            byte[] answerData =
                    answer.getBytes();

            DatagramPacket answerPacket =
                    new DatagramPacket(
                            answerData,
                            answerData.length,
                            receivePacket.getAddress(),
                            receivePacket.getPort()
                    );

            socket.send(answerPacket);

            System.out.println(
                    "Đã gửi kết quả: " + answer
            );

            // ==============================
            // 8. Đóng socket
            // ==============================

            socket.close();

            System.out.println("Đã đóng socket.");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }

package org.example;


import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;

public class udp_string {

    public static void main(String[] args) {

        String serverIP = "36.50.135.242";
        int serverPort = 2208;

        String studentCode = "B22DCCN200";
        String qCode = "og08BQex";

        try {
            // 1. Tạo UDP socket
            DatagramSocket socket = new DatagramSocket();

            InetAddress serverAddress =
                    InetAddress.getByName(serverIP);

            // 2. Gửi mã sinh viên + mã câu hỏi
            String request =
                    ";" + studentCode + ";" + qCode;

            byte[] sendData =
                    request.getBytes("UTF-8");

            DatagramPacket sendPacket =
                    new DatagramPacket(
                            sendData,
                            sendData.length,
                            serverAddress,
                            serverPort
                    );

            socket.send(sendPacket);

            System.out.println("Đã gửi: " + request);

            // 3. Nhận dữ liệu từ server
            byte[] receiveData = new byte[4096];

            DatagramPacket receivePacket =
                    new DatagramPacket(
                            receiveData,
                            receiveData.length
                    );

            socket.receive(receivePacket);

            String response =
                    new String(
                            receivePacket.getData(),
                            0,
                            receivePacket.getLength(),
                            "UTF-8"
                    );

            System.out.println("Server gửi: " + response);

            // 4. Tách requestId và data
            String[] parts = response.split(";", 2);

            String requestId = parts[0];
            String data = parts[1];

            // 5. Đếm số lần xuất hiện của từng ký tự
            int[] count = new int[65536];

            for (int i = 0; i < data.length(); i++) {

                char c = data.charAt(i);

                count[c]++;
            }

            // 6. Tìm ký tự xuất hiện nhiều nhất
            char maxChar = data.charAt(0);
            int maxCount = count[maxChar];

            for (int i = 1; i < data.length(); i++) {

                char c = data.charAt(i);

                if (count[c] > maxCount) {
                    maxCount = count[c];
                    maxChar = c;
                }
            }

            // 7. Tạo kết quả
            StringBuilder answer =
                    new StringBuilder();

            answer.append(requestId);
            answer.append(";");
            answer.append(maxChar);
            answer.append(":");

            // 8. Tìm vị trí ký tự
            // QUAN TRỌNG: vị trí bắt đầu từ 1
            for (int i = 0; i < data.length(); i++) {

                if (data.charAt(i) == maxChar) {

                    answer.append(i + 1);
                    answer.append(",");
                }
            }

            // 9. Gửi kết quả
            String result = answer.toString();

            System.out.println("Kết quả: " + result);

            byte[] answerData =
                    result.getBytes("UTF-8");

            DatagramPacket answerPacket =
                    new DatagramPacket(
                            answerData,
                            answerData.length,
                            receivePacket.getAddress(),
                            receivePacket.getPort()
                    );

            socket.send(answerPacket);

            // 10. Đóng socket
            socket.close();

            System.out.println("Đã đóng socket.");

        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

            // =========================================

                    new StringBuilder();

                    normalized.append(" ");

            );

            // =========================================
            // 6. Tạo thông điệp gửi về server

            // requestId;data
            // =========================================

                    requestId + ";" + resultData;

                    "Kết quả gửi: " + answer
            );

            // =========================================

            byte[] answerData =

package org.example;

import UDP.Student;

import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.ObjectInputStream;
import java.io.ObjectOutputStream;
import java.net.DatagramPacket;
import java.net.DatagramSocket;
import java.net.InetAddress;
import java.nio.charset.StandardCharsets;

public class udp_object1 {

    static final String SERVER = "36.50.135.242";
    static final int PORT = 2209;

    static final String STUDENT_CODE = "B22DCCN200";
    static final String QUESTION_CODE = "IepfQ48M";

    // Chuẩn hóa họ tên:
    // nguyen van tuan nam
    // -> Nguyen Van Tuan Nam
    static String normalizeName(String name) {

        name = name.trim().replaceAll("\\s+", " ");

        String[] words = name.split(" ");

        StringBuilder result = new StringBuilder();

        for (String word : words) {

            if (word.length() == 0) {
                continue;
            }

            word = word.toLowerCase();

            String normalized =
                    Character.toUpperCase(word.charAt(0))
                            + word.substring(1);

            if (result.length() > 0) {
                result.append(" ");
            }

            result.append(normalized);
        }

        return result.toString();
    }

    // Tạo username:
    // Nguyen Van Tuan Nam
    // -> namnvt
    //
    // Lấy tên cuối cùng + chữ cái đầu của các từ phía trước.
    static String createUsername(String name) {

        String[] words = name.toLowerCase().split("\\s+");

        if (words.length == 0) {
            return "";
        }

        StringBuilder username = new StringBuilder();

        // Tên
        username.append(words[words.length - 1]);

        // Chữ cái đầu của họ và tên đệm
        for (int i = 0; i < words.length - 1; i++) {
            username.append(words[i].charAt(0));
        }

        return username.toString();
    }

    // Chuyển Student thành mảng byte để gửi qua UDP
    static byte[] serializeStudent(Student student) throws Exception {

        ByteArrayOutputStream bos = new ByteArrayOutputStream();

        ObjectOutputStream oos = new ObjectOutputStream(bos);

        oos.writeObject(student);
        oos.flush();

        return bos.toByteArray();
    }

    // Chuyển mảng byte nhận được thành Student
    static Student deserializeStudent(byte[] data, int offset, int length)
            throws Exception {

        ByteArrayInputStream bis =
                new ByteArrayInputStream(data, offset, length);

        ObjectInputStream ois = new ObjectInputStream(bis);

        return (Student) ois.readObject();
    }

    public static void main(String[] args) {

        try (DatagramSocket socket = new DatagramSocket()) {

            InetAddress serverAddress =
                    InetAddress.getByName(SERVER);

            /*
             * a. Gửi:
             * ;B22DCCN200;IepfQ48M
             */
            String request = ";" + STUDENT_CODE + ";" + QUESTION_CODE;

            byte[] requestData =
                    request.getBytes(StandardCharsets.UTF_8);

            DatagramPacket requestPacket =
                    new DatagramPacket(
                            requestData,
                            requestData.length,
                            serverAddress,
                            PORT
                    );

            socket.send(requestPacket);

            System.out.println("Da gui: " + request);

            /*
             * b. Nhận dữ liệu từ server.
             *
             * 8 byte đầu: requestId
             * Các byte còn lại: đối tượng Student
             */
            byte[] buffer = new byte[65507];

            DatagramPacket responsePacket =
                    new DatagramPacket(buffer, buffer.length);

            socket.receive(responsePacket);

            int length = responsePacket.getLength();

            // Lấy 8 byte đầu
            String requestId =
                    new String(
                            buffer,
                            0,
                            8,
                            StandardCharsets.UTF_8
                    );

            System.out.println("Request ID: " + requestId);

            // Deserialize Student từ byte thứ 8 trở đi
            Student student =
                    deserializeStudent(
                            buffer,
                            8,
                            length - 8
                    );

            System.out.println("Student nhan duoc:");
            System.out.println(student);

            /*
             * c1. Chuẩn hóa họ tên
             */
            String normalizedName =
                    normalizeName(student.getName());

            student.setName(normalizedName);

            /*
             * c2. Tạo email
             *
             * Ví dụ:
             * Nguyen Van Tuan Nam
             * -> namnvt@ptit.edu.vn
             */
            String username =
                    createUsername(student.getName());

            String email =
                    username + "@ptit.edu.vn";

            student.setEmail(email);

            System.out.println("Student sau khi xu ly:");
            System.out.println(student);

            /*
             * c3. Serialize Student đã sửa
             */
            byte[] studentData =
                    serializeStudent(student);

            /*
             * Tạo packet mới:
             *
             * 8 byte requestId
             * +
             * byte[] Student
             */
            byte[] requestIdData =
                    requestId.getBytes(StandardCharsets.UTF_8);

            byte[] sendData =
                    new byte[8 + studentData.length];

            // 8 byte đầu
            System.arraycopy(
                    requestIdData,
                    0,
                    sendData,
                    0,
                    8
            );

            // Student từ byte thứ 8
            System.arraycopy(
                    studentData,
                    0,
                    sendData,
                    8,
                    studentData.length
            );

            DatagramPacket sendPacket =
                    new DatagramPacket(
                            sendData,
                            sendData.length,
                            serverAddress,
                            PORT
                    );

            socket.send(sendPacket);

            System.out.println("Da gui Student da xu ly len server.");

        } catch (Exception e) {

            e.printStackTrace();
        }
    }
}


 package org.example;

 import UDP.Customer;

 import java.io.ByteArrayInputStream;
 import java.io.ByteArrayOutputStream;
 import java.io.ObjectInputStream;
 import java.io.ObjectOutputStream;
 import java.net.DatagramPacket;
 import java.net.DatagramSocket;
 import java.net.InetAddress;
 import java.nio.charset.StandardCharsets;

 public class udp_object2 {

     private static final String SERVER = "36.50.135.242";
     private static final int PORT = 2209;

     private static final String STUDENT_CODE = "B22DCCN200";
     private static final String QUESTION_CODE = "HQx7FkxP";


     // =========================================================
     // 1. Chuẩn hóa tên
     //
     // Input:
     // nguyen van hai duong
     //
     // Output:
     // DUONG, Nguyen Van Hai
     // =========================================================
     public static String normalizeName(String name) {

         // Xóa khoảng trắng đầu/cuối
         // và gom nhiều khoảng trắng thành một
         name = name.trim().replaceAll("\\s+", " ");

         String[] words = name.split(" ");

         StringBuilder result = new StringBuilder();

         // Lấy tên cuối
         String lastName = words[words.length - 1].toUpperCase();

         result.append(lastName);
         result.append(", ");

         // Lấy họ + tên đệm
         for (int i = 0; i < words.length - 1; i++) {

             String word = words[i].toLowerCase();

             word = Character.toUpperCase(word.charAt(0))
                     + word.substring(1);

             if (i > 0) {
                 result.append(" ");
             }

             result.append(word);
         }

         return result.toString();
     }


     // =========================================================
     // 2. Tạo username
     //
     // Input:
     // nguyen van hai duong
     //
     // Output:
     // nvhduong
     // =========================================================
     public static String createUserName(String name) {

         name = name.trim().replaceAll("\\s+", " ");

         String[] words = name.split(" ");

         StringBuilder username = new StringBuilder();

         // Lấy chữ cái đầu của họ và tên đệm
         for (int i = 0; i < words.length - 1; i++) {

             username.append(
                     Character.toLowerCase(words[i].charAt(0))
             );
         }

         // Lấy toàn bộ tên cuối
         username.append(
                 words[words.length - 1].toLowerCase()
         );

         return username.toString();
     }


     // =========================================================
     // 3. Đổi ngày sinh
     //
     // Input:
     // 10-11-2012
     //
     // Output:
     // 11/10/2012
     //
     // mm-dd-yyyy -> dd/mm/yyyy
     // =========================================================
     public static String formatDate(String date) {

         String[] parts = date.split("-");

         String month = parts[0];
         String day = parts[1];
         String year = parts[2];

         return day + "/" + month + "/" + year;
     }


     // =========================================================
     // 4. Serialize Customer
     // Object -> byte[]
     // =========================================================
     public static byte[] serialize(Customer customer)
             throws Exception {

         ByteArrayOutputStream byteArrayOutputStream =
                 new ByteArrayOutputStream();

         ObjectOutputStream objectOutputStream =
                 new ObjectOutputStream(byteArrayOutputStream);

         objectOutputStream.writeObject(customer);
         objectOutputStream.flush();

         return byteArrayOutputStream.toByteArray();
     }


     // =========================================================
     // 5. Deserialize Customer
     // byte[] -> Object
     // =========================================================
     public static Customer deserialize(
             byte[] data,
             int offset,
             int length
     ) throws Exception {

         ByteArrayInputStream byteArrayInputStream =
                 new ByteArrayInputStream(
                         data,
                         offset,
                         length
                 );

         ObjectInputStream objectInputStream =
                 new ObjectInputStream(byteArrayInputStream);

         return (Customer) objectInputStream.readObject();
     }


     // =========================================================
     // MAIN
     // =========================================================
     public static void main(String[] args) {

         try {

             DatagramSocket socket = new DatagramSocket();

             InetAddress serverAddress =
                     InetAddress.getByName(SERVER);


             // =================================================
             // BƯỚC 1
             // Gửi:
             // ;B22DCCN200;HQx7FkxP
             // =================================================

             String request =
                     ";" + STUDENT_CODE + ";" + QUESTION_CODE;

             byte[] requestData =
                     request.getBytes(StandardCharsets.UTF_8);

             DatagramPacket requestPacket =
                     new DatagramPacket(
                             requestData,
                             requestData.length,
                             serverAddress,
                             PORT
                     );

             socket.send(requestPacket);

             // =================================================
             // BƯỚC 2
             // Nhận dữ liệu từ server
             //
             // 8 byte đầu: requestId
             // Các byte còn lại: Customer
             // =================================================

             byte[] buffer = new byte[65507];

             DatagramPacket responsePacket =
                     new DatagramPacket(
                             buffer,
                             buffer.length
                     );

             socket.receive(responsePacket);

             int responseLength =
                     responsePacket.getLength();


             // Lấy 8 byte requestId
             String requestId =
                     new String(
                             buffer,
                             0,
                             8,
                             StandardCharsets.UTF_8
                     );

             System.out.println(
                     "Request ID: " + requestId
             );


             // Lấy Customer từ byte thứ 8
             Customer customer =
                     deserialize(
                             buffer,
                             8,
                             responseLength - 8
                     );

             System.out.println(
                     "Customer nhan duoc:"
             );

             System.out.println(customer);


             // =================================================
             // BƯỚC 3
             // Chuẩn hóa tên
             //
             // nguyen van hai duong
             // ->
             // DUONG, Nguyen Van Hai
             // =================================================

             String newName =
                     normalizeName(
                             customer.getName()
                     );

             customer.setName(newName);


             // =================================================
             // BƯỚC 4
             // Đổi ngày sinh
             //
             // 10-11-2012
             // ->
             // 11/10/2012
             // =================================================

             String newDate =
                     formatDate(
                             customer.getDayOfBirth()
                     );

             customer.setDayOfBirth(newDate);


             // =================================================
             // BƯỚC 5
             // Tạo username
             //
             // Dùng tên ban đầu để tạo username:
             //
             // nguyen van hai duong
             // ->
             // nvhduong
             // =================================================

             /*
              * Vì sau khi normalizeName(), name đã trở thành:
              *
              * DUONG, Nguyen Van Hai
              *
              * nên cần lấy lại phần tên theo định dạng mới
              * để tạo username.
              */

             String normalizedName =
                     customer.getName();

             String[] nameParts =
                     normalizedName.split(", ");

             String firstMiddleNames =
                     nameParts[1];

             String[] words =
                     firstMiddleNames.split(" ");

             String lastName =
                     nameParts[0].toLowerCase();

             StringBuilder username =
                     new StringBuilder();

             // Chữ cái đầu của họ + tên đệm
             for (String word : words) {

                 username.append(
                         Character.toLowerCase(
                                 word.charAt(0)
                         )
                 );
             }

             // Tên cuối
             username.append(lastName);

             customer.setUserName(
                     username.toString()
             );


             // =================================================
             // Hiển thị Customer sau khi xử lý
             // =================================================

             System.out.println(
                     "\nCustomer sau khi xu ly:"
             );

             System.out.println(customer);


             // =================================================
             // BƯỚC 6
             // Serialize Customer
             // =================================================

             byte[] customerData =
                     serialize(customer);


             // =================================================
             // BƯỚC 7
             // Tạo packet:
             //
             // 8 byte requestId
             // +
             // Object Customer
             // =================================================

             byte[] requestIdData =
                     requestId.getBytes(
                             StandardCharsets.UTF_8
                     );

             byte[] sendData =
                     new byte[
                             8 + customerData.length
                             ];


             // Copy requestId vào 8 byte đầu
             System.arraycopy(
                     requestIdData,
                     0,
                     sendData,
                     0,
                     8
             );


             // Copy Customer vào phía sau
             System.arraycopy(
                     customerData,
                     0,
                     sendData,
                     8,
                     customerData.length
             );


             // =================================================
             // BƯỚC 8
             // Gửi lại server
             // =================================================

             DatagramPacket sendPacket =
                     new DatagramPacket(
                             sendData,
                             sendData.length,
                             serverAddress,
                             PORT
                     );

             socket.send(sendPacket);

             System.out.println(
                     "\nDa gui Customer da xu ly len server."
             );


             // =================================================
             // BƯỚC 9
             // Đóng socket
             // =================================================

             socket.close();

             System.out.println(
                     "Da dong socket."
             );

         } catch (Exception e) {

             e.printStackTrace();
         }
     }
 }
